        // ==========================================
        // AI Challenge Generator Logic

        const challenges = [
            "Today's AI Challenge: Go completely paperless for all your notes today! 📄➡️💻",
            "Today's AI Challenge: Persuade one friend to switch to a reusable water bottle. 💧",
            "Today's AI Challenge: Unplug all unused chargers and electronics in your room. 🔌",
            "Today's AI Challenge: Audit your classroom's energy use and switch off empty lights. 💡",
            "Today's AI Challenge: Choose public transport or bike instead of a car today! 🚌🚴",
            "Today's AI Challenge: Eat a plant-based meal for lunch today. 🥗",
            "Today's AI Challenge: Identify 3 items to recycle at home and sort them properly. ♻️",
            "Today's AI Challenge: Plant a native species or care for an existing plant. 🌱",
            "Today's AI Challenge: Reduce your screen time and spend 30 minutes outdoors. 🌳",
            "Today's AI Challenge: Check product labels and choose sustainable options. 🏷️"
        ];

        function generateChallenge() {
            const randomIdx = Math.floor(Math.random() * challenges.length);
            document.getElementById("challengeText").innerText = challenges[randomIdx];
        }

        // ==========================================
        // Game 1: Trivia Card Logic
        // ==========================================
        const triviaData = [
            { q: "Which gas is primarily responsible for global climate warming?", options: ["Oxygen", "Carbon Dioxide", "Nitrogen"], correct: 1 },
            { q: "Roughly how long does a standard plastic bottle take to decompose?", options: ["10 years", "50 years", "450+ years"], correct: 2 },
            { q: "What is a major benefit of digital minimalism for the environment?", options: ["Saves server energy & reduces e-waste", "Cleans oceans directly", "Stops rain"], correct: 0 },
            { q: "How much CO2e can you save annually by eating vegetarian?", options: ["100 kg", "500 kg", "1000 kg"], correct: 1 },
            { q: "What is the primary cause of plastic pollution?", options: ["Recycling failures", "Single-use plastics", "Ocean currents"], correct: 1 },
            { q: "Which activity reduces your carbon footprint the most?", options: ["Taking fewer flights", "Using LED bulbs", "Recycling"], correct: 0 }
        ];

        let currentTrivia = 0;
        let scoreTrivia = 0;

        function loadTrivia() {
            if (currentTrivia < triviaData.length) {
                let data = triviaData[currentTrivia];
                document.getElementById("question").innerText = `Q${currentTrivia + 1}: ${data.q}`;
                let optionsHtml = "";
                data.options.forEach((opt, index) => {
                    optionsHtml += `<button onclick="checkTrivia(${index})">${opt}</button>`;
                });
                document.getElementById("options").innerHTML = optionsHtml;
                document.getElementById("triviaScore").innerText = `Score: ${scoreTrivia}/${triviaData.length}`;
            } else {
                const percentage = Math.round((scoreTrivia / triviaData.length) * 100);
                document.getElementById("triviaBox").innerHTML = `<div class="completion-message">🎉 Trivia Completed!<br>Final Score: ${scoreTrivia}/${triviaData.length} (${percentage}%)</div>`;
            }
        }

        function checkTrivia(selected) {
            if (selected === triviaData[currentTrivia].correct) {
                scoreTrivia++;
            }
            currentTrivia++;
            loadTrivia();
        }

        // ==========================================
        // Game 2: Quick Casual Sorting Arcade Logic
        // ==========================================
        const items = [
            { name: "Plastic Bottle", type: "recycle" },
            { name: "Banana Peel", type: "trash" },
            { name: "Cardboard Box", type: "recycle" },
            { name: "Used Tissue", type: "trash" },
            { name: "Aluminum Soda Can", type: "recycle" },
            { name: "Coffee Grounds", type: "trash" },
            { name: "Glass Jar", type: "recycle" },
            { name: "Plastic Bag", type: "recycle" },
            { name: "Paper Towel", type: "trash" },
            { name: "Steel Can", type: "recycle" }
        ];

        let currentItemIndex = 0;
        let arcadeScore = 0;

        function nextItem() {
            currentItemIndex = Math.floor(Math.random() * items.length);
            document.getElementById("itemToSwipe").innerText = `Item: ${items[currentItemIndex].name}`;
        }

        function sortItem(userChoice) {
            if (items[currentItemIndex].type === userChoice) {
                arcadeScore += 10;
                document.getElementById("itemToSwipe").style.color = "#2e7d32";
            } else {
                arcadeScore = Math.max(0, arcadeScore - 5);
                document.getElementById("itemToSwipe").style.color = "#d32f2f";
            }
            document.getElementById("arcadeScore").innerText = `Score: ${arcadeScore}`;
            setTimeout(() => {
                document.getElementById("itemToSwipe").style.color = "#1b5e20";
                nextItem();
            }, 300);
        }

        // Initialize on page load
        loadTrivia();
        nextItem();
    </script>
